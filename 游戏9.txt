// ========== 本地存储读写 ==========
function loadUserData(){
  const saved = localStorage.getItem("charGameUserData");
  if(saved) userData = JSON.parse(saved);
  document.getElementById("soundCheck").checked = userData.soundOn;
  document.getElementById("speakCheck").checked = userData.speakOn;
  renderRank();
}
function saveUserData(){
  localStorage.setItem("charGameUserData", JSON.stringify(userData));
  renderRank();
}
function getRank(exp){
  let curRank = rankList[0];
  for(let r of rankList){
    if(exp >= r.exp) curRank = r;
  }
  return curRank;
}
function renderRank(){
  const r = getRank(userData.exp);
  const el = document.getElementById("rankText");
  if(el) el.innerText = `当前称号：${r.title}｜累计经验：${userData.exp}`;
}

// ========== 音效 ==========
function playSound(isCorrect){
  if(!userData.soundOn) return;
  const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.connect(gain); gain.connect(audioCtx.destination);
  if(isCorrect){
    osc.frequency.value = 520;
    gain.gain.value = 0.25;
    osc.start();
    setTimeout(()=>osc.stop(),180);
  }else{
    osc.frequency.value = 220;
    gain.gain.value =0.25;
    osc.start();
    setTimeout(()=>osc.stop(),250);
  }
}
//文字朗读
function speakText(txt){
  if(!userData.speakOn) return;
  const utter = new SpeechSynthesisUtterance(txt);
  utter.lang = "zh-CN";
  speechSynthesis.speak(utter);
}

  goAddWord();
}

// 获取对应模式的答案文本
function getAnswerByMode(item, mode){
  if(mode === "pinyin") return item.py;
  if(mode === "group") return item.group;
  if(mode === "chengyu") return item.idiom;
  return item.py;

// 开始普通答题游戏
function startGame(){
  setting.timer = document.getElementById("timerCheck").checked;
  const allWords = getAllWordList();
  let filterWords = allWords.filter(item=>item.diff === setting.diff);
  filterWords.sort(()=>Math.random()-0.5);
  gameState.questionPool = filterWords.slice(0, setting.qCount);
  if(gameState.questionPool.length === 0){
    alert("当前难度下没有题库，请补充汉字！");
    return;
  }
  gameState.currentIndex = 0;
  gameState.score = 0;
  gameState.isWrongReview = false;
  showQuestion();
  showPage("page-quiz");
}

//渲染普通答题
function showQuestion(){
  clearTimeout(gameState.timerObj);
  const item = gameState.questionPool[gameState.currentIndex];
  let questionTxt, answerKey, realMode;
  if(setting.mode === "pinyin"){
    questionTxt = item.hz;
    answerKey = item.py;
    realMode = "pinyin";
  }else if(setting.mode === "group"){
    questionTxt = item.hz;
    answerKey = item.group;
    realMode = "group";
  }else if(setting.mode === "chengyu"){
    questionTxt = item.hz;
    answerKey = item.idiom;
    realMode = "chengyu";
  }else{
    const rand = Math.floor(Math.random()*3);
    if(rand===0){questionTxt=item.hz;answerKey=item.py;realMode="pinyin";}
    else if(rand===1){questionTxt=item.hz;answerKey=item.group;realMode="group";}
    else{questionTxt=item.hz;answerKey=item.idiom;realMode="chengyu";}
  }
  speakText(questionTxt);
  let wrongOptions = [];
  const pool = [...gameState.questionPool];
  while(wrongOptions.length<3 && pool.length>1){
    const randWord = pool[Math.floor(Math.random()*pool.length)];
    let opt;
    if(realMode === "pinyin") opt = randWord.py;
    else if(realMode === "group") opt = randWord.group;
    else opt = randWord.idiom;
    if(opt!==answerKey && !wrongOptions.includes(opt)) wrongOptions.push(opt);
  }
  let allOptions = [answerKey, ...wrongOptions];
  allOptions.sort(()=>Math.random()-0.5);
  document.getElementById("progressText").innerText = `第${gameState.currentIndex+1}题 / 共${gameState.questionPool.length}题`;
  let html = `<div class="question-char">${questionTxt}</div>`;
  allOptions.forEach(opt=>{
    html += `<button class="option-btn">${opt}</button>`
  })
  document.getElementById("quiz-content").innerHTML = html;
  //绑定选项点击事件
  const optList = allOptions;
  const itemIndex = gameState.currentIndex;
  document.querySelectorAll("#quiz-content .option-btn").forEach((btn, idx)=>{
    btn.onclick = ()=> checkAnswer(optList[idx], answerKey, btn, itemIndex, realMode);
  })

  if(setting.timer){
    gameState.timerObj = setTimeout(()=>{
      const exist = userData.wrongList.some(w=>w.hz === item.hz);
      if(!exist){
        userData.wrongList.push({...item, wrongMode:realMode});
        saveUserData();
      }
      nextQuestion();
    }, 20000);
  }
}

//渲染错题复习
function showWrongQuestion(){
  clearTimeout(gameState.timerObj);
  const item = gameState.questionPool[gameState.currentIndex];
  const realMode = item.wrongMode || "pinyin";
  const questionTxt = item.hz;
  const answerKey = getAnswerByMode(item, realMode);
  speakText(questionTxt);

  let wrongOptions = [];
  const pool = [...gameState.questionPool];
  while(wrongOptions.length<3 && pool.length>1){
    const randWord = pool[Math.floor(Math.random()*pool.length)];
    let opt = getAnswerByMode(randWord, realMode);
    if(opt!==answerKey && !wrongOptions.includes(opt)) wrongOptions.push(opt);
  }
  let allOptions = [answerKey, ...wrongOptions];
  allOptions.sort(()=>Math.random()-0.5);

  document.getElementById("wrongReviewTip").innerText = `错题复习 ${gameState.currentIndex+1}/${gameState.questionPool.length}｜模式：${realMode}`;
  let html = `<div class="question-char">${questionTxt}</div>`;
  allOptions.forEach(opt=>{
    html += `<button class="option-btn">${opt}</button>`
  })
  document.getElementById("wrongQuizContent").innerHTML = html;
  //修复：把选项数组放到外层，点击回调可以正常读取
  const optList = allOptions;
  const poolIndex = gameState.currentIndex;
  document.querySelectorAll("#wrongQuizContent .option-btn").forEach((btn, i)=>{
    btn.onclick = ()=> checkWrongReview(optList[i], poolIndex, btn);
  })
}

// 判断答案 正常答题
function checkAnswer(userAns, realAns, btnDom, itemIndex, realMode){
  clearTimeout(gameState.timerObj);
  const wordItem = gameState.questionPool[itemIndex];
  const allBtns = document.querySelectorAll("#quiz-content .option-btn");
  allBtns.forEach(b=>b.disabled=true);
  if(userAns === realAns){
    gameState.score +=1;
    userData.exp +=10;
    btnDom.classList.add("answer-correct");
    playSound(true);
    showStarAnim(false);
    const idx = userData.wrongList.findIndex(w=>w.hz === wordItem.hz);
    if(idx>-1) userData.wrongList.splice(idx,1);
  }else{
    btnDom.classList.add("answer-wrong");
    playSound(false);
    allBtns.forEach(b=>{if(b.innerText === realAns) b.classList.add("answer-correct");})
    const exist = userData.wrongList.some(w=>w.hz === wordItem.hz);
    if(!exist) userData.wrongList.push({...wordItem, wrongMode:realMode});
  }
  saveUserData();
  setTimeout(nextQuestion,1200);
}

//错题复习答题
function checkWrongReview(userAns, poolIndex, btnDom){
  clearTimeout(gameState.timerObj);
  const wordItem = gameState.questionPool[poolIndex];
  const allBtns = document.querySelectorAll("#page-wrongreview .option-btn");
  allBtns.forEach(b=>b.disabled=true);
  const realMode = wordItem.wrongMode || "pinyin";
  const realAns = getAnswerByMode(wordItem, realMode);

  if(userAns === realAns){
    btnDom.classList.add("answer-correct");
    playSound(true);
    showStarAnim(true);
    //答对，删除原始错题
    const idx = userData.wrongList.findIndex(w=>w.hz === wordItem.hz);
    if(idx>-1) userData.wrongList.splice(idx,1);
    userData.exp +=8;
  }else{
    btnDom.classList.add("answer-wrong");
    playSound(false);
    allBtns.forEach(b=>{if(b.innerText === realAns) b.classList.add("answer-correct");})
    const exist = userData.wrongList.some(w=>w.hz === wordItem.hz);
    if(!exist) userData.wrongList.push({...wordItem, wrongMode: realMode});
  }
  saveUserData();
  setTimeout(nextWrongQuestion,1200);
}

//普通模式下一题
function nextQuestion(){
  gameState.currentIndex +=1;
  if(gameState.currentIndex >= gameState.questionPool.length){
    document.getElementById("result-content").innerHTML = `
      <p>答题完成！</p>
      <p>得分：${gameState.score} / ${gameState.questionPool.length}</p>
      <p>正确率：${(gameState.score/gameState.questionPool.length*100).toFixed(1)}%</p>
      <p>本次获得经验：${gameState.score*10}</p>
    `;
    showPage("page-result");
  }else{
    showQuestion();
  }
}
//错题复习下一题
function nextWrongQuestion(){
  gameState.currentIndex +=1;
  if(gameState.currentIndex >= gameState.questionPool.length){
    alert("错题复习完成！已经清除答对的题目。");
    backHome();
  }else{
    showWrongQuestion();
